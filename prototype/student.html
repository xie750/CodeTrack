Node* deleteAt(Node* head, int pos) {
    if (head == NULL) return head;
    Node* cur = head;
    for (int i = 1; i < pos - 1 && cur != NULL; i++) {
        cur = cur->next;
    }
    if (cur != NULL && cur->next != NULL) {
        Node* target = cur->next;
        cur->next = target->next;
        delete target;
    }
    return head;
}

为什么链表删除头节点容易出错？
